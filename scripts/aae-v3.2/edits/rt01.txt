test('R6: gate command waits for a READY result on the current source',async t=>{const {root}=setup(t,{c:cmd({purpose:'publish',gate:'ready'})});prompt(root,'AAE GO');await assert.rejects(runCommand(root,'test'),/Gate test: RESULT\.json ontbreekt/);put(root,RESULT,{...reportTemplate(root)});await assert.rejects(runCommand(root,'test'),/niet READY/);assert.equal(state(root).task.usage.commands,0);put(root,RESULT,evidence(root));const r=await runCommand(root,'test');assert.equal(r.exit_code,0);});
test('R6: gate refuses a READY result from an older source',async t=>{const {root}=setup(t,{c:cmd({purpose:'publish',gate:'ready'})});prompt(root,'AAE GO');put(root,RESULT,evidence(root));put(root,'src/a.js','export const a=7;');await assert.rejects(runCommand(root,'test'),/Broncode veranderd/);});
test('R12: in-scope change to a watched script keeps the command approved',async t=>{const {root}=setup(t,{c:cmd({max_runs:2}),extra:{scope:{read:['src'],write:['src/a.js','check.mjs']}}});prompt(root,'AAE GO');assert.equal((await runCommand(root,'test')).exit_code,0);put(root,'check.mjs','console.log("aangepast binnen scope");');assert.equal((await runCommand(root,'test')).exit_code,0);});
test('B4: in-scope change to a watched file still needs a new GO for publish commands',async t=>{const {root}=setup(t,{c:cmd({purpose:'publish',max_runs:2}),extra:{scope:{read:['src'],write:['src/a.js','check.mjs']}}});prompt(root,'AAE GO');put(root,'check.mjs','console.log("anders");');await assert.rejects(runCommand(root,'test'),/niet goedgekeurd/);});
test('B4: in-scope change to a watched file still needs a new GO for a local gate command',async t=>{const {root}=setup(t,{c:cmd({purpose:'test',gate:'ready',max_runs:2}),extra:{scope:{read:['src'],write:['src/a.js','check.mjs']}}});prompt(root,'AAE GO');put(root,'check.mjs','console.log("anders");');await assert.rejects(runCommand(root,'test'),/niet goedgekeurd/);});
test('R4: amended route keeps approved commands runnable without a new GO',async t=>{const {root,config}=setup(t,{c:cmd({max_runs:2})});prompt(root,'AAE GO');const res=installTask(root,{...config,goal:'Zelfde doel met een aangescherpte beschrijving binnen de scope.'});assert.equal(res.status,'active');assert.equal((await runCommand(root,'test')).exit_code,0);});
test('R3: follow-up prompt keeps command approval',async t=>{const {root}=setup(t,{c:cmd({max_runs:2})});prompt(root,'AAE GO');prompt(root,'Testresultaat: alles ok.');assert.equal((await runCommand(root,'test')).exit_code,0);});
test('R5: publish can repeat within its approved run limit but is never trusted',async t=>{const c=cmd({purpose:'publish',max_runs:3});const {root}=setup(t,{c});prompt(root,'AAE GO');prompt(root,'AAE VERTROUW');assert.equal(state(root).trusted_commands[commandFingerprint(root,c)],undefined);for(let i=0;i<3;i++)assert.equal((await runCommand(root,'test')).exit_code,0);await assert.rejects(runCommand(root,'test'),/budget bereikt|Maximum/);});
